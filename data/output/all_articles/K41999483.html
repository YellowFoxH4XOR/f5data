Description

You can use the script in this article to troubleshoot integrity check issues on a BIG-IP system licensed with the FIPS 140-2 option, testing either the active or any inactive software volume.

Environment

  • BIG-IP hardware appliance
  • The BIG-IP device has a license that includes the FIPS 140-2 option (FIPS full-box license)

Recommended Actions

  1. Copy the script below (or download it) and save it to the BIG-IP.
    Note: On a VIPRION with multiple blades, consider saving the script to somewhere such as /config or /root, and the system will copy the file to all blades.

  2. Make the script executable by running chmod +x

    chmod +x /root/check-sys-eicheck.sh && touch /root/check-sys-eicheck.sh

  3. Run the script and specify the name of a particular software volume, e.g. running it against the software in HD1.3:

    /root/check-sys-eicheck.sh HD1.3

    Note: On a VIPRION, you can use "clsh" to run the script on every blade.  See K16727: Overview of the cluster shell utility for more information

Script contents:

#!/bin/bash

set -o errexit
set -o errtrace
set -o pipefail
set -o nounset
# set -o xtrace
shopt -s nullglob

err_handler()
{
    echo "Error ($?) on line $1"
}
trap 'err_handler $LINENO' ERR

FIPSERR_FILES=(
    "/config/f5_public/fipserr"
    "/var/named/config/f5_public/fipserr"
    "/var/dnscached/config/f5_public/fipserr"
)

mount_volumeset()
{
    local set="$1"
    local target="$2"

    local prefix=""
    for i in /dev/mapper/vg--*-set."$set".root; do
        if [ -n "$prefix" ]; then
            echo >&2 "Warning: script doesn't support RAID"
            return 1
        fi
        prefix="$(basename "$i" ".$set.root")"
    done

    if [ -z "$prefix" ]; then
        echo >&2 "Warning: didn't find software volume set $set"
        return 1
    fi

    mount -o ro "/dev/mapper/$prefix.$set.root" "$target"
    mount -o ro "/dev/mapper/$prefix.$set._config" "$target/config"
    mount -o ro "/dev/mapper/$prefix.$set._usr" "$target/usr"
    mount -o ro "/dev/mapper/$prefix.$set._var" "$target/var"

    for i in dev proc sys var/run; do
        mount -o bind "/$i" "$target/$i"
        mount -o remount,bind,ro "$target/$i"
    done
    if [ -d "$target/run" ]; then
        # mount /var/run to /run as well, to handle pre-v14.1 to v14.1+ transitions
        mount -o bind "/var/run" "$target/run"
        mount -o remount,bind,ro "$target/run"
    fi
}

unmount_volumeset()
{
    local target="$1"
    # Find all the mount points, reverse the order (so that the target itself
    # is last), and pass to umount.
    findmnt -l -n -o TARGET --submount "$target" | tac | \
        xargs --no-run-if-empty umount
}

cleanup_volumeset_mount()
{
    local target="$1"
    # lazy attempt to mitigate race conditions with things still holding the
    # chroot filesystems open
    sleep 1

    unmount_volumeset "$target" || true
    rmdir "$target" || true
}

check_fipserr()
{
    echo "Checking /config/f5_public/fipserr..."

    local target="$1"

    for file in "${FIPSERR_FILES[@]}"; do
        if [ -s "$target$file" ]; then
            echo "WARNING: $file in target software volume has contents (power-on test will/may fail):"
            cat "$target$file"
            echo
        fi
    done
}

check_sys_eicheck()
{
    echo "Checking sys-eicheck..."

    local target="$1"
    local result

    if ! result="$(chroot "$target" /usr/libexec/sys-eicheck.py)"; then
        if [ "$target" = "/" ]; then
            echo "WARNING: sys-eicheck failed."
        else
            cat <<EOF
WARNING: sys-eicheck failed. This may be a false positive (because checking
a non-running software volume) if the failures are only 'missing critical
files' under /var/run and /run:
EOF
        fi

        cat <<EOF
===
$result
===
EOF

# An example of a false-positive:
# =======================================================
# missing     /var/run/console
# missing     /var/run/nslcd
# missing     /var/run/console
# missing     /run/httpd
#
# 4 critical file(s) missing
#
# This is partially mitigated by mounting /run and /var/run in the chroot,
# but things such as 'nslcd' will still trip up the check

    fi
}

big3d_version()
{
    local binary="$1"
    (strings -a "$binary" || true) | grep -m 1 "big3d Version"
}

check_big3d_version()
{
    echo "Checking big3d version..."

    local target="$1"

    # If either file doesn't exist, do nothing (although it's concerning that
    # big3d doesn't exist in the target volume, that should be picked up by
    # sys-eicheck.py if it's an issue).
    if ! [ -r "$target/usr/sbin/big3d" ] || ! [ -r /shared/bin/big3d ]; then
        return 0
    fi

    local global_version target_version
    global_version="$(big3d_version /shared/bin/big3d)"
    target_version="$(big3d_version "$target/usr/sbin/big3d")"

    # This checks is imperfect, but errs on the side of issuing a warning.
    # The more complex version would need to check mtime and parse the version
    # strings.
    if [ "$global_version" != "$target_version" ]; then
        local remove_cmd="rm -f /shared/bin/big3d"
        if [ "$CURRENT_CHASSIS_SLOT" -gt 0 ]; then
            remove_cmd="clsh $remove_cmd"
        fi

        cat <<EOF
WARNING: big3d version differs between /shared/bin/big3d and target software
volume! Power-on self test may fail! Remove /shared/bin/big3d before rebooting
into new software volume ('$remove_cmd')

Current /shared/bin/big3d: $global_version
Target software volume's big3d: $target_version
EOF
    fi
}


if [ $# -lt 1 ]; then
    cat >&2 <<EOF
Usage: $0 <volume set>

e.g. $0 HD1.1
EOF
    exit 1
fi

# HD1.3 => 3
TARGET_VOLUME="${1##*.}"
CURRENT_VOLUME="$(switchboot -c)"
CURRENT_VOLUME="${CURRENT_VOLUME##*.}"
CURRENT_CHASSIS_SLOT="$(getdb platform.chassis.slot)"

if [ "$CURRENT_CHASSIS_SLOT" -gt 0 ]; then
    echo "Running on blade $CURRENT_CHASSIS_SLOT"
fi

if [ "$TARGET_VOLUME" = "$CURRENT_VOLUME" ]; then
    echo "Running against current boot location..."
    check_fipserr /
    check_sys_eicheck /
    check_big3d_version /
else
    echo "Running against other boot location ($TARGET_VOLUME)"
    TMP_MOUNT="$(mktemp -d)"
    if [ ! -d "$TMP_MOUNT" ]; then
        echo >&2 "Warning: failed to create tempdir"
        exit 1
    fi

    trap 'cleanup_volumeset_mount "$TMP_MOUNT"' EXIT
    mount_volumeset "$TARGET_VOLUME" "$TMP_MOUNT"
    check_fipserr "$TMP_MOUNT"
    check_sys_eicheck "$TMP_MOUNT"
    check_big3d_version "$TMP_MOUNT"
    trap - EXIT

    unmount_volumeset "$TMP_MOUNT"
    rmdir "$TMP_MOUNT"
fi

echo 'All done!'
