Description

  • Enable Cross-Origin Resource Sharing (CORS) to a specific VirtualServer in Nginx Ingress Controller.

Environment

  • Nginx Ingress Controller

Cause

None.

 Recommended Actions

1. Run the command below and add the option, '-enable-snippets', to enable SNIPPETS.

 #kubectl edit deployments.apps -n nginx-ingress nginx-ingress
-----
apiVersion: apps/v1
kind: Deployment
metadata:
 annotations:
...
 name: nginx-ingress
 namespace: nginx-ingress
spec:
 selector:
   matchLabels:
     app: nginx-ingress
...
 template:
   metadata:
     labels:
       app: nginx-ingress
   spec:
     automountServiceAccountToken: true
     containers:
     - args:
       - -enable-snippets

 

2. Run the command below and add the CORS-related location-snippets configuration to the desired path.

#kubectl edit vs -n <namespace> <virtual-server-name>
----
apiVersion: k8s.nginx.org/v1
kind: VirtualServer
metadata:
 annotations:
...
 name: <VirtualServer-Name>
 namespace: default
spec:
 host: <VirtualServer-Hostname>
 routes:
 - action:
     pass: <Upstream-Name>
   path: <Desired-Path>
   location-snippets: |
     if ($request_method = 'OPTIONS') {
       add_header 'Access-Control-Allow-Origin' '*';
       add_header 'Access-Control-Allow-Methods' 'GET, POST, OPTIONS';
       add_header 'Access-Control-Allow-Headers' 'DNT,User-Agent,X-Requested-With,If-Modified-Since,Cache-Control,Content-Type,Range';
       add_header 'Access-Control-Max-Age' 1728000;
       add_header 'Content-Type' 'text/plain; charset=utf-8';
       add_header 'Content-Length' 0;
       return 204;
     }
     if ($request_method = 'POST') {
       add_header 'Access-Control-Allow-Origin' '*' always;
       add_header 'Access-Control-Allow-Methods' 'GET, POST, OPTIONS' always;
       add_header 'Access-Control-Allow-Headers' 'DNT,User-Agent,X-Requested-With,If-Modified-Since,Cache-Control,Content-Type,Range' always;
       add_header 'Access-Control-Expose-Headers' 'Content-Length,Content-Range' always;
     }
     if ($request_method = 'GET') {
       add_header 'Access-Control-Allow-Origin' '*' always;
       add_header 'Access-Control-Allow-Methods' 'GET, POST, OPTIONS' always;
       add_header 'Access-Control-Allow-Headers' 'DNT,User-Agent,X-Requested-With,If-Modified-Since,Cache-Control,Content-Type,Range' always;
       add_header 'Access-Control-Expose-Headers' 'Content-Length,Content-Range' always;
     }
 upstreams:
 - name: <Upstream-Name>
   port: <Backend-Port>
   service: <Backend-Service>

 

3. Check on the Nginx Config via the command below to see if Access-Controll-* headers-related configurations are there under the desired location.

#kubectl exec -n nginx-ingress <nginx-ingress-pod-name> -it -- nginx -T
