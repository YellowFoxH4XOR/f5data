Description

When you create a topology the iRules that are built as a result log locally by default. In order to change this alterations to these iRules will be needed.

Environment
  • You have created a topology and need to log remotely.
  • SSL Profiles, Virtual servers, pool members, HSL logging, Bypass IPs/URLs

Cause

This could be due to the volume of logging being created or a local decision to do so.

Recommended Actions

High Speed Logging is the best way to accomplish this. In order to do this the changes will have to be made manually to the iRule.

  1. For example to send logging for a Bypass rule in the topology there are certain lines that would need to be added and/or modified.

    Here is an example of a typical Bypass iRule created for a common topology.
ltm rule /Common/Topology_Select_Prod {
    when RULE_INIT {
    ## User-defined: array list of to-match categories for alternate egress
    set static::ALT_PATH_URLS {
        /Common/ZScaler_URL_Bypass_List
    }

    ## User-defined: data group of destination subnet to-match
    set static::ALT_PATH_SUBNETS "wccp_bypass_subnets"

    ## User-defined: name of alternate path topology
    set static::ALT_TOPOLOGY "/Common/sslo_Zscaler_Bypass_Prod.app/sslo_Zscaler_Bypass_Prod-in-t-4"

    ## User-defined: name of primary path topology
    set static::PRIMARY_TOPOLOGY "/Common/sslo_Zscaler_Path_Prod.app/sslo_Zscaler_Path_Prod-in-t-4"
}
when CLIENT_ACCEPTED {

    if {[class match -- [IP::local_addr] equals $static::ALT_PATH_SUBNETS ]} {
        virtual $static::ALT_TOPOLOGY
        log local0.info "F5_SSLO_Log Topology=Zscaler_Bypass_Prod Proxied=No Path=Zscaler_Bypass F5_Rule=ZS_Bypassed_subnets src_ip=[IP::remote_addr] dest_ip=[IP::local_addr] dest_port=[TCP::local_port]"
        ## disable remaining event processing
        event CLIENT_DATA disable
        event CLIENTSSL_CLIENTHELLO disable
    } else {
        virtual $static::PRIMARY_TOPOLOGY
        TCP::collect
    }
}
when CLIENT_DATA {
    if { [regexp {^([A-Z]+) (?:/|https?://|[A-Za-z\d-]{1,63}[.]|\[[a-fA-F\d:]+\]|(?:[*]\x20)) } [TCP::payload] junk method] && ([lsearch -exact -sorted {CONNECT DELETE GET HEAD LOCK OPTIONS PATCH POST PROPFIND PUT TRACE UNLOCK} $method] >= 0 )} {
        set host_hdr [findstr [TCP::payload] "Host: " 6 "\r\n"]
        set res [CATEGORY::lookup "http://${host_hdr}/" custom]
        if { [lsearch -exact $static::ALT_PATH_URLS ${res}] >= 0 } {
            virtual $static::ALT_TOPOLOGY
            log local0.info "F5_SSLO_Log Topology=Zscaler_Bypass_Prod Proxied=No Path=Zscaler_Bypass F5_Rule=ZS_Bypassed_URLs cat=${res} url=http://${host_hdr}/ src_ip=[IP::remote_addr] dest_ip=[IP::local_addr] dest_port=[TCP::local_port]"
            ## disable remaining event processing
            event CLIENTSSL_CLIENTHELLO disable
        } else { log local0.info "F5_SSLO_Log Topology=Zscaler_Path_Prod Proxied=Yes Path=Zscaler F5_Rule=None url=http://${host_hdr}/ src_ip=[IP::remote_addr] dest_ip=[IP::local_addr] dest_port=[TCP::local_port]"}
    }
    TCP::release
}
when CLIENTSSL_CLIENTHELLO {
    if { [SSL::extensions exists -type 0] } {
        set tls_servername [string range [SSL::extensions -type 0] 9 [string length [SSL::extensions -type 0]]]
        if { [info exists tls_servername] } {
            set res [CATEGORY::lookup "https://${tls_servername}/" custom]
            if { [lsearch -exact $static::ALT_PATH_URLS ${res}] >= 0 } {
                log local0.info "F5_SSLO_Log Topology=Zscaler_Bypass_Prod Proxied=No Path=Direct_to_Internet F5_Rule=ZS_Bypassed_URLs cat=${res} url=https://${tls_servername}/ src_ip=[IP::remote_addr] dest_ip=[IP::local_addr] dest_port=[TCP::local_port]"
                virtual $static::ALT_TOPOLOGY
            } else { log local0.info "F5_SSLO_Log Topology=Zscaler_Path_Prod Proxied=Yes Path=Zscaler F5_Rule=None url=https://${tls_servername}/ src_ip=[IP::remote_addr] dest_ip=[IP::local_addr] dest_port=[TCP::local_port]" }
        }
    } else { log local0.info "F5_SSLO_Log Topology=Zscaler_Path_Prod Proxied=Yes Path=ZScaler F5_Rule=None/No_SNI url=no_SNI src_ip=[IP::remote_addr] dest_ip=[IP::local_addr] dest_port=[TCP::local_port]" }
}
}
