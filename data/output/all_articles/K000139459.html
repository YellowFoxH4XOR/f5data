Description

This article outlines how to add the Show Password feature via the Advanced Customizations features of APM. 
Modern Customization - added JS will add a checkbox to show the password at the Login page and also when the user needs to confirm password changes.

Environment

  • BIG-IP APM
  • Standard or Modern Advanced Customizations
  • APM Login Page

Cause

This is not a feature that is currently offered by APM.

Recommended Actions

A. ) Using Modern Customization
Note: If the user-logon.js file becomes unworkable, you can copy the contents from an unmodified user-logon.js file and replace it with the non-working code.

To add a checkbox for Show Password - you will need to update the user-logon.js file to include the following code.

  1. Login to the BIG-IP GUI
  2. Navigate to: Access  ››  Profiles / Policies : Customization : Advanced
  3. Expand the Access Profile for the login page you would like to modify.
    Access Policy --> Logon Pages --> Logon Page --> user-logon.js

  1. Add the following code to the Advanced Customization Editor of the user-login.js file.

    Object.defineProperty(exports, "__esModule", { value: true });
    requirejs.config({
        map: {
            'apmui/master/View': {
                'apmui/page/logon/View': module.id,
            },
        },
    });
 
    var CustomLogonView = /** @class */ (function (_super) {
        tslib_1.__extends(CustomLogonView, _super);
        function CustomLogonView() {
            return _super !== null && _super.apply(this, arguments) || this;
        }
 
        CustomLogonView.prototype.componentDidMount = function () {
            _super.prototype.componentDidMount.call(this);
 
            // Add checkbox for original password field
            var passwordDiv = document.querySelector('#password-container');
            if (passwordDiv) {
                passwordDiv.insertAdjacentHTML('afterend', '<div><input type="checkbox" onclick="window.togglePassword()"> Show Password</div>');
 
                window.togglePassword = function () {
                    var x = document.getElementById("password");
                    if (x && x.type === "password") {
                        x.type = "text";
                    } else if (x) {
                        x.type = "password";
                    }
                };
            } else {
                console.error('Password container not found!');
            }
 
            // Add show password checkbox for new fields
            function addShowPasswordCheckbox() {
                var verifyPasswordDiv = document.getElementById('_F5_verify_password-container');
                if (verifyPasswordDiv) {
                    verifyPasswordDiv.insertAdjacentHTML('afterend', '<div><input type="checkbox" id="show-password-checkbox"> Show Password</div>');
 
                    var checkbox = document.getElementById('show-password-checkbox');
                    if (checkbox) {
                        checkbox.addEventListener('click', function () {
                            togglePasswordVisibility('_F5_challenge');
                            togglePasswordVisibility('_F5_verify_password');
                        });
                    }
                } else {
                    console.error('Password container for _F5_verify_password not found!');
                }
            }
 
            // Function to toggle password visibility
            function togglePasswordVisibility(fieldId) {
                var inputField = document.getElementById(fieldId);
                if (inputField && inputField.type === "password") {
                    inputField.type = "text";
                } else if (inputField) {
                    inputField.type = "password";
                }
            }
 
            // Add show password for both fields
            addShowPasswordCheckbox();
        };
