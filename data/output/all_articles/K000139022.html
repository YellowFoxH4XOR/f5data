Description

Self IP addresses in TMOS are sorted first by whichever object was last modified.

Environment

  • BIG-IP device
  • creating self ip addresses or modifying existing self ip addresses

Cause

Objects are listed in alphanumerical order in the corresponding /config/*.conf file or when executing the "tmsh list" command in the CLI.

In the bigip_base.conf file, "net self" objects are ordered by whichever self ip was modified last.

root@(localhost1)(cfg-sync In Sync)(Standby)(/Common)(tmos)# list net self
net self 10.2.115.170/16 {
    address 10.2.115.170/16
    allow-service all
    traffic-group traffic-group-local-only
    vlan Internal
}
net self 10.79.111.99/16 {
    address 10.79.111.99/16
    allow-service all
    floating enabled
    traffic-group traffic-group-1
    unit 1
    vlan External
}
net self 10.2.115.172/16 {
    address 10.2.115.172/16
    allow-service all
    floating enabled
    traffic-group traffic-group-1
    unit 1
    vlan Internal
}
net self 10.79.111.97/16 {
    address 10.79.111.97/16
    allow-service all
    traffic-group traffic-group-local-only
    vlan External
}

# create new self ip
root@(localhost1)(cfg-sync In Sync)(Standby)(/Common)(tmos)# create net self zself address 10.90.111.100/16 vlan Internal

root@(localhost1)(cfg-sync In Sync)(Standby)(/Common)(tmos)# list net self
net self zself {      <<<< new self ip is ordered first
    address 10.90.111.100/16
    traffic-group traffic-group-local-only
    vlan Internal
}
net self 10.2.115.170/16 {
    address 10.2.115.170/16
    allow-service all
    traffic-group traffic-group-local-only
    vlan Internal
}
net self 10.79.111.99/16 {
    address 10.79.111.99/16
    allow-service all
    floating enabled
    traffic-group traffic-group-1
    unit 1
    vlan External
}
net self 10.2.115.172/16 {
    address 10.2.115.172/16
    allow-service all
    floating enabled
    traffic-group traffic-group-1
    unit 1
    vlan Internal
}
net self 10.79.111.97/16 {
    address 10.79.111.97/16
    allow-service all
    traffic-group traffic-group-local-only
    vlan External
}

# modify an existing self ip address
root@(localhost1)(cfg-sync In Sync)(Standby)(/Common)(tmos)# modify net self 10.79.111.97/16 allow-service none

root@(localhost1)(cfg-sync In Sync)(Standby)(/Common)(tmos)# list net self
net self 10.79.111.97/16 {     <<<< the modified self ip address is now listed first
    address 10.79.111.97/16
    traffic-group traffic-group-local-only
    vlan External
}
net self zself {
    address 10.90.111.100/16
    traffic-group traffic-group-local-only
    vlan Internal
}
net self 10.2.115.170/16 {
    address 10.2.115.170/16
    allow-service all
    traffic-group traffic-group-local-only
    vlan Internal
}
net self 10.79.111.99/16 {
    address 10.79.111.99/16
    allow-service all
    floating enabled
    traffic-group traffic-group-1
    unit 1
    vlan External
}
net self 10.2.115.172/16 {
    address 10.2.115.172/16
    allow-service all
    floating enabled
    traffic-group traffic-group-1
    unit 1
    vlan Internal
}

# Note that the "show net self" output still orders the objects alphanumerically, not by last modified date
root@(localhost1)(cfg-sync Changes Pending)(Standby)(/Common)(tmos)# show net self | grep ^Net::Self
Net::Self IP: 10.2.115.170_16
Net::Self IP: 10.2.115.172_16
Net::Self IP: 10.79.111.97_16
Net::Self IP: 10.79.111.99_16
Net::Self IP: zself

Recommended Actions

None. BIG-IP orders "net self" objects in the /config/bigip_base.conf file by the last modified date.

Additional Information

All other objects in /config/*.conf are ordered alphanumerically. 

For example, "net route" objects are ordered alphanumerically, and does not match the order each object was last modified.

# create
root@(localhost1)(cfg-sync Changes Pending)(Standby)(/Common)(tmos)# create net route 10.10.1.0/24  interface Internal network 10.10.1.0/24
root@(localhost1)(cfg-sync Changes Pending)(Standby)(/Common)(tmos)# create net route zstatic_route3 interface Internal network 10.10.30.0/24
root@(localhost1)(cfg-sync Changes Pending)(Standby)(/Common)(tmos)# create net route static_route1  interface Internal network 10.10.10.0/24
root@(localhost1)(cfg-sync Changes Pending)(Standby)(/Common)(tmos)# create net route zstatic_route2 interface Internal network 10.10.200.0/24

# Static routes are ordered alphanumerically.
root@(localhost1)(cfg-sync Changes Pending)(Standby)(/Common)(tmos)# list net route
net route 10.10.1.0/24 {
    interface /Common/Internal
    network 10.10.1.0/24
}
net route static_route1 {
    interface /Common/Internal
    network 10.10.10.0/24
}
net route zstatic_route2 {
    interface /Common/Internal
    network 10.10.200.0/24
}
net route zstatic_route3 {
    interface /Common/Internal
    network 10.10.30.0/24
}

# Note the "show net route" table orders them by the creation date.
root@(localhost1)(cfg-sync Changes Pending)(Standby)(/Common)(tmos)# show net route static
