Description

When you run tmsh list ltm profile client-ssl command, "_0" is added to the end of cert-key-chain name.  

Environment

BIG-IP LTM with client SSL profiles configured

Cause

This is behavior change introduced in ID 535122
Below is an example configuration for versions prior to the change. 
  • New behavior
[root@bigip:Active:Standalone] config # tmsh list ltm profile client-ssl Test-Client-Profile
ltm profile client-ssl Test-Client-Profile {
    alert-timeout indefinite
    allow-dynamic-record-sizing disabled
    allow-non-ssl disabled
    app-service none
    cache-size 262144
    cache-timeout 3600
    cert-key-chain {
        default_0 {                     <== add "_0"
            cert default.crt
            key default.key
        }
    }
    inherit-certkeychain false
    key bigip-web.key
    passphrase none
}
  • Old behavior

Before BIG-IP V14.x there is no letter added after cert-key-chain name
[root@bigip:Active:Standalone] config # tmsh list ltm profile client-ssl Test-Client-Profile
ltm profile client-ssl nan_clientssl_test {
  alert-timeout indefinite
    allow-dynamic-record-sizing disabled
    allow-non-ssl disabled
    app-service none
    cache-size 262144
    cache-timeout 3600
    cert-key-chain {
        default {                     <=== There is no "_0" added to the configuration 
            cert default.crt
            key default.key
        }
    }
    inherit-certkeychain false
    key bigip-web.key
    passphrase none
}
