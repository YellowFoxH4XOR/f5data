Description

A BIG-IP licensed for AFM has two layers of local self IP protection. There is the normal BIG-IP local self IP allow service and an AFM notwork policy. in an OSPF environment both BIG-IP allow service and AFM network policy have to be applied to the self IP for OSPF to function. If both are not set correctly then the following issues will be noticed:

 

  • Neighbors stuck exchanging DBDs
  • Neighbors stuck in Init state
  • BIG-IP side not seeing Neighbors

Environment

  • BIG-IP licensed for AFM
  • OSPF routing configured

 

Cause

Local self IP config does not allow traffic through,neither does it allow traffic through AFM network firewall. There is a default allow service for local self IPs which allows specific default ports but does not apply in AFM firewall context.

Recommended Actions

Carry out steps.

  1. Create a security policy based on security firewall rule-list _sys_self_allow_defaults
  2. Attach the security policy to the local self IP
  3. Attach default allow-service to the local self IP.

Additional Information

[root@ospf_fw01:Standby:In Sync] config # tmsh list security firewall rule-list _sys_self_allow_defaults

security firewall rule-list _sys_self_allow_defaults {

  rules {

    _sys_allow_tcp_defaults {

      action accept

      ip-protocol tcp

      destination {

        port-lists {

          _sys_self_allow_tcp_defaults

        }

      }

    }

    _sys_allow_udp_defaults {

      action accept

      ip-protocol udp

      destination {

        port-lists {

          _sys_self_allow_udp_defaults

        }

      }

    }

    _sys_allow_ospf_defaults {

      action accept

      ip-protocol ospf

    }

    _sys_allow_pim_defaults {

      action accept

      ip-protocol pim

    }

    _sys_allow_igmp_defaults {

      action accept

      ip-protocol igmp

    }

  }

}

 

[root@ospf_fw01:Standby:In Sync] config # tmsh list security firewall port-list _sys_self_allow_tcp_defaults

security firewall port-list _sys_self_allow_tcp_defaults {

  ports {

    1029-1043 { }

    domain { }

    f5-iquery { }

    https { }

    snmp { }

    ssh { }

  }

}

 

[root@ospf_fw01:Standby:In Sync] config # tmsh list security firewall port-list _sys_self_allow_udp_defaults

security firewall port-list _sys_self_allow_udp_defaults {

  ports {

    520 { }

    cap { }

    domain { }

    f5-iquery { }

    snmp { }

  }

}

 

[root@ospf_fw01:Standby:In Sync] config # tmsh create security firewall policy fw_self_ip_default_allow rules add { _Common__sys_self_allow_defaults { rule-list _sys_self_allow_defaults place-before first } }

 

[root@ospf_fw01:Standby:In Sync] config # tmsh list security firewall policy fw_self_ip_default_allow

security firewall policy fw_self_ip_default_allow {

  description "Defaults for AFM self allow"

  rules {

    _Common__sys_self_allow_defaults {

      rule-list _sys_self_allow_defaults

    }

  }

}

 

[root@ospf_fw01:Standby:In Sync] config # tmsh modify net self OSPF_SELF_LOCAL fw-enforced-policy fw_self_ip_default_allow allow-service default

 

[root@ospf_fw02:Active:In Sync] config # tmsh modify net route-domain 0 { fw-enforced-policy none }; tmsh save sys config

 

[root@ospf_fw01:Standby:In Sync] config # tmsh list net self OSPF_SELF_LOCAL

net self OSPF_SELF_LOCAL {

  address 192.0.2.6/29

  allow-service {

    default

  }

  fw-enforced-policy fw_self_ip_default_allow

  traffic-group traffic-group-local-only

  vlan OSPF_VLAN

}
