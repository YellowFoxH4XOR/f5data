K000138638: AS3 error Declaration failed: 0107082a:3: All objects must be removed from a partition

Type
Support Solution
Published
2024-02-16
Updated
2024-02-16
Applies to
  • F5 Automation and Orchestration
  • F5 Application Services 3 Extension
Source
https://my.f5.com/manage/s/article/K000138638

Description

AS3 error "Declaration failed: 0107082a:3: All objects must be removed from a partition (services) before the partition may be removed, type ID (28726)","level":"error" When trying to delete a Tenant

 

Environment

  • AS3
  • Partition
  • POSTing declaration to BIG-IP

Cause

 

Config added to the tenant manually after previously posting config via AS3

Recommended Actions

 

To resolve the issue you need to POST a dummy config  on the same tenant "services" via AS3. Then you will need to comment the config which was manually added to the partition.
Example: Dummy AS3 config below

{
   "class": "AS3",
   "action": "deploy",
   "persist": true,
   "declaration": {
      "class": "ADC",
      "schemaVersion": "3.0.0",
      "id": "urn:uuid:33045210-3ab8-4636-9b2a-c98d22ab915d",
      "label": "Sample 1",
      "remark": "Simple HTTP application with RR pool",
      "services": {
         "class": "Tenant",
         "A1": {
            "class": "Application",
            "service": {
               "class": "Service_HTTP",
               "virtualAddresses": [
                  "10.1.1.113"
               ],
               "pool": "web_pool"
            },
            "web_pool": {
               "class": "Pool",
               "monitors": [
                  "http"
               ],
               "members": [{
                  "servicePort": 80,
                  "serverAddresses": [
                     "172.10.10.10",
                     "172.10.10.11"
                  ]
               }]
            }
         }
      }
   }
}

Now after the above dummy AS3 config is successfully POSTed on the BIG-IP, you will need to go to the bigip.conf file on the partition services and edit the file via vi editor and comment the lines as shown below:

Note: In the below config we have created a bot profile (/services/example-bot-profile) manually in the partition services

#TMSH-VERSION: 14.1.5.3

ltm pool /services/A1/web_pool {
    members {
        /Common/172.10.10.10:80 {
            address 172.10.10.10
            description "(replaces AS3 172.10.10.10)"
        }
        /Common/172.10.10.11:80 {
            address 172.10.10.11
            description "(replaces AS3 172.10.10.11)"
        }
    }
    min-active-members 1
    monitor min 1 of { /Common/http }
}

ltm virtual /services/A1/service {
    creation-time 2024-02-06:09:31:51
    description A1
    destination /services/10.1.1.113:80
    ip-protocol tcp
    last-modified-time 2024-02-06:09:31:51
    mask 255.255.255.255
    persist {
        /Common/cookie {
            default yes
        }
    }
    pool /services/A1/web_pool
    profiles {
        /Common/f5-tcp-progressive { }
        /Common/http { }
    }
    source 0.0.0.0/0
    source-address-translation {
        type automap
    }
    translate-address enabled
    translate-port enabled
}

ltm virtual-address /services/10.1.1.113 {
    address 10.1.1.113
    arp enabled
    icmp-echo enabled
    inherited-traffic-group true
    mask 255.255.255.255
    traffic-group /Common/traffic-group-1
}

#security bot-defense asm-profile /services/example-bot-profile {
#    app-service none
#}


Then save the config file.

Once done run the command

tmsh load sys config verify

If the comand does not return any errors then you can run the command

tmsh load sys config

tmsh save sys config

 

After this DELETE the config via AS3 and now you will be able to remove the partition.
