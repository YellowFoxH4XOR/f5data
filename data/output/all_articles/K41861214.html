Description

An iRule for debugging HTTPS traffic passing through a Standard virtual server (VIP).

Environment

  • Standard virtual server (VIP)
  • Configured with Client and Server SSL profiles
  • iRules

Cause

None

Recommended Actions

Add the following iRule to the virtual server:
Note:  Replace <CLIENT_IP> with the Test Client's IP address

when CLIENT_ACCEPTED {
  if { [clientside {IP::remote_addr}] == "<CLIENT_IP>" } {
    set Debug_Enable true
    log local0.debug "[clientside {IP::remote_addr}]:[clientside {TCP::remote_port}] [clock clicks -milliseconds]"
    TCP::collect
  } else {
    set Debug_Enable false
  }
}

when CLIENT_DATA {
  log local0.debug "[clientside {IP::remote_addr}]:[clientside {TCP::remote_port}] [clock clicks -milliseconds]: L4C>S [TCP::payload length] [b64encode [TCP::payload 512]]"
  TCP::release
}

when HTTP_REQUEST {
  if { [info exists Debug_Enable] && $Debug_Enable } {
    log local0.debug "[clientside {IP::remote_addr}]:[clientside {TCP::remote_port}] [clock clicks -milliseconds]: L7C>S Hd [HTTP::method] [HTTP::uri]"
      foreach aHeader [HTTP::header names] {
        log local0.debug "[clientside {IP::remote_addr}]:[clientside {TCP::remote_port}] [clock clicks -milliseconds]: L7C>S Hd $aHeader: [HTTP::header values $aHeader]"
      }
      log local0.debug "[clientside {IP::remote_addr}]:[clientside {TCP::remote_port}] [clock clicks -milliseconds]: L7C>S Bd [b64encode [HTTP::payload 512]]"
      log local0.debug "[clientside {IP::remote_addr}]:[clientside {TCP::remote_port}] [clock clicks -milliseconds]: --== ## ==--"
    }
  }

when HTTP_REQUEST_SEND {
  if { [info exists Debug_Enable] && $Debug_Enable } {
    log local0.debug "[clientside {IP::remote_addr}]:[clientside {TCP::remote_port}] [clock clicks -milliseconds]: -> balanced to -> [serverside {IP::remote_addr}]:[serverside {TCP::remote_port}]"
    TCP::collect
  }
}

when SERVER_DATA {
  log local0.debug "[clientside {IP::remote_addr}]:[clientside {TCP::remote_port}] [clock clicks -milliseconds]: <- balancing from <- [serverside {IP::remote_addr}]:[serverside {TCP::remote_port}]"
  log local0.debug "[clientside {IP::remote_addr}]:[clientside {TCP::remote_port}] [clock clicks -milliseconds]: L4C<S [TCP::payload length] [b64encode [TCP::payload 512]]"
  TCP::release
}

when HTTP_RESPONSE {
  if { [info exists Debug_Enable] && $Debug_Enable } {
    log local0.debug "[clientside {IP::remote_addr}]:[clientside {TCP::remote_port}] [clock clicks -milliseconds]: L7C<S Hd [HTTP::status]"
      foreach aHeader [HTTP::header names] {
      log local0.debug "[clientside {IP::remote_addr}]:[clientside {TCP::remote_port}] [clock clicks -milliseconds]: L7C<S Hd $aHeader: [HTTP::header value $aHeader]"
    }
    log local0.debug "[clientside {IP::remote_addr}]:[clientside {TCP::remote_port}] [clock clicks -milliseconds]: L7C<S Bd [b64encode [HTTP::payload 512]]"
    log local0.debug "[clientside {IP::remote_addr}]:[clientside {TCP::remote_port}] [clock clicks -milliseconds]: --== ## ==--"
  }
}
