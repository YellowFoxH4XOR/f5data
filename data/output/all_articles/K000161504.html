Description

After upgrading to BIG-IP v17.x, you may encounter the following error message from a previously issued iRule for Edge Client Functionality

EdgeClient_Default_Browser_Auth:49: warning: [unmatched closing character][}]

Environment

  • BIG-IP APM v17.x 
  • Edge Client Compatibility
  • iRules

Cause

Additional TCL/JSON parsing resolved. 

Answer/Recommended Actions

Replace the iRule with the following for Modern Customization for both Windows and Mac:

#Default browser Authentication for windows Edge client with Auto Epi

when CLIENT_ACCEPTED {
    ACCESS::restrict_irule_events disable
}

when HTTP_REQUEST {
    if { [HTTP::uri] contains "vdesk/webtop.eui?" } {
        set session_id  [ACCESS::session sid]
        set landinguri [ACCESS::session data get -sid $session_id session.server.landinguri]
        if { $landinguri contains "?EC=1&port=" } {
            set redirect_port [URI::query $landinguri port]
            set state_param [URI::query $landinguri state]
            set otc -1
            set trimsid [string range $session_id 24 32]
            table set $trimsid 1 604800
            log local0. "on edgclient EC, rendered updated webtop and add/modify entry $trimsid"

            if { $landinguri contains "os=macos" } {
                # do nothing and render the default webtop
                log local0. "on edgclient EC MAC, doing nothing"
            }
            else {
                set html_response "<html><head><title>Connecting to Edge client.</title><link rel=\"stylesheet\" href=\"/public/include/css/modern/framework.css\" /></head>"
                append html_response "<body><div id=\"apmui-container\" class=\"apmui-container\"><div class=\"apmui-root apmui-page-policy apmui-page-logout\">"
                append html_response "<header class=\"apmui-header\"><div class=\"apmui-header--content\"><picture class=\"apmui-header--logo\">"
                append html_response "<source media=\"(max-width: 586px)\" srcset=\"/public/images/modern/general/f5-logo-mobile.png\">"
                append html_response "<img src=\"/public/images/modern/general/f5-logo-desktop.png\" alt=\"F5 Logo\"></picture><span></span></div></header><main class=\"apmui-main\">"
                append html_response "<div class=\"apmui-content\"><h3 id=\"responseText\">Connecting to server...</h3>"
                append html_response "<span style=\"display: none;\" id=\"otc_code\">$otc</span><span style=\"display: none;\" id=\"_port\">$redirect_port</span>"
                append html_response "<span style=\"display: none;\" id=\"_state\">$state_param</span></div></main><footer class=\"apmui-footer\">"
                append html_response "<div class=\"apmui-footer--content\" id=\"copyright\">This product is licensed from F5, Inc. (c) 1999-2024 F5, Inc. All rights reserved.</div></footer></div></div>"
                append html_response "<script>function getCookieValue(cookieName) \{var name = cookieName + \"=\"; var decodedCookie = decodeURIComponent(document.cookie);"
                append html_response "var cookieArray = decodedCookie.split(\';\'); for (var i = 0; i < cookieArray.length; i++) \{"
                append html_response "var cookie = cookieArray\[i\].trim(); if (cookie.indexOf(name) === 0) \{ return cookie.substring(name.length, cookie.length); \}"
                append html_response "\} return null;\} var F5_ST = getCookieValue(\'F5_ST\'); var TIN = getCookieValue(\'TIN\');"
                append html_response "const url = new URL(window.location.href); const hostname = url.protocol + \"//\" + url.hostname + (url.port ? \":\" + url.port : \"\");"
                append html_response "let logoutUrl =  url.protocol + \"//\" + url.hostname + \"/my.logout.php3?hangup_error=1\" ;"
                append html_response "function closeTab() \{setTimeout(() => \{window.close();\}, \"5000\");\}"
                append html_response "function sendReqToLocalhost(port, otcCode, state) \{const xhr = new XMLHttpRequest(); xhr.open(\"GET\", \"http://localhost:\"+ port + \"?code=\" + otcCode + \"&f5_st=\" + F5_ST + \"&tin=\" + TIN + \"&host=\" + hostname + \"&state=\" + state, true);"
                append html_response "xhr.send(null); xhr.onreadystatechange = function () \{ if (xhr.readyState === XMLHttpRequest.DONE && xhr.status === 200) \{"
                append html_response "document.getElementById(\"responseText\").innerHTML=\"You can close this page now.\";\} else if (xhr.readyState === XMLHttpRequest.DONE && xhr.status !== 200) \{"
                append html_response "document.getElementById(\"responseText\").innerHTML = \"Something went wrong, Communication with server failed.\"; window.location.href= logoutUrl;\}\}\}"
                append html_response "function makeServerRequest() \{var otcCode = document.getElementById(\"otc_code\").textContent;"
                append html_response "var port = document.getElementById(\"_port\").textContent;"
                append html_response "var state = document.getElementById(\"_state\").textContent;"
                append html_response "const xhr = new XMLHttpRequest(); xhr.open(\"GET\", \"/vdesk/get_token_for_sessid.php3\", true);"
                append html_response "xhr.send(null); xhr.onreadystatechange = function () \{"
                append html_response "if (xhr.readyState === XMLHttpRequest.DONE && xhr.status === 200) \{"
                append html_response "otcCode = xhr.getResponseHeader(\'X-ACCESS-Session-Token\'); sendReqToLocalhost(port, otcCode, state);"
                append html_response "\} else if (xhr.readyState === XMLHttpRequest.DONE && xhr.status !== 200) \{"
                append html_response "sendReqToLocalhost(port, otcCode, state); document.getElementById(\"responseText\").innerHTML = \"Something went wrong, getting token for sessid.\";"
                append html_response "\}\}\}makeServerRequest();</script></body></html>"
                HTTP::respond 200 content $html_response
            }
        }
    }
    elseif { [HTTP::uri] contains "vdesk/hangup.php3" } {
        #hangup due to access policy failures
        if { [ACCESS::session exists -state_deny] } {
            set session_id  [ACCESS::session sid]
            set landinguri [ACCESS::session data get -sid $session_id session.server.landinguri]
            if { $landinguri contains "?EC=1&port=" } {
                set redirect_port [URI::query $landinguri port]
                set state_param [URI::query $landinguri state]
                if {[info exists redirect_port] and [info exists state_param]} {
                    log local0. "Access policy deny"
                    set hangupReq 1
                }
            }
        }
    }
    elseif { [HTTP::uri] contains "my.policy_host" } {
        set session_id  [ACCESS::session sid]
        set landinguri [ACCESS::session data get -sid $session_id session.server.landinguri]
        if { $landinguri contains "?EC=1&port=" } {
            set cookieString [HTTP::header value "Cookie"]
            set cookie_parts [split $cookieString ";"]
            set filtered_cookie ""
            foreach part $cookie_parts {
                if {[string match "*MachineHash*" $part] == 0} {
                    append filtered_cookie "$part;"
                }
            }
            if {[string length $filtered_cookie] > 0} {
                set filtered_cookie [string trimright $filtered_cookie ";"]
            }
            HTTP::header replace "Cookie" $filtered_cookie
        }
    }
    elseif { [HTTP::uri] contains "public/include/js/modern/main.js" } {
        # main.js specfically is for auto click on epi handling
        # whereas main.js triggers in logout too so only update on valid session
        set session_id  [ACCESS::session sid]
        if { $session_id != ""} {
            set landinguri [ACCESS::session data get -sid $session_id session.server.landinguri]
            if { $landinguri contains "?EC=1&port=" } {

                if { $landinguri contains "os=macos" } {
                    # serve mac_main.js for mac edgeclient cases
                    set contentValue [ifile get mac_main.js]
                    log local0. "on webvpn/embedded, rendered mac_main.js"
                    HTTP::respond 200 content $contentValue "Cache-Control" "no-store"
                }
                else {
                    set redirect_port [URI::query $landinguri port]
                    set state_param [URI::query $landinguri state]
                    set port $redirect_port
                    set contentValue [ifile get main.js]
                    set originalString "http://localhost:8000/epi/?args="
                    set newString "http://localhost:$port/epi/?state=$state_param&args="
                    set newjs [string map [ list $originalString $newString] $contentValue]
                    set trimsid [string range $session_id 24 32]
                    table set $trimsid 1 604800
                    log local0. "on edgclient EC, rendered updated main.js and add/modify entry $trimsid"
                    HTTP::respond 200 content $newjs  "Cache-Control" "no-store"
                }
            }
            else {
                #defaultmain js file for webvpn cases and tells browser to cache no-store
                set contentValue [ifile get defaultmain.js]
                log local0. "on webvpn/embedded, rendered defaultmain.js"
                HTTP::respond 200 content $contentValue "Cache-Control" "no-store"
            }
        }
        else {
            # on no session disconnect/logout case;
            # where default main.js triggers which will be memory cached in browser that caused the awaiting state due to epi is missing.
            set apm_cookie [HTTP::cookie value LastMRH_Session]
            if { $apm_cookie != ""} {
                set edgBro [table lookup -notouch $apm_cookie]
                if { $edgBro != "" } {
                    set contentValue [ifile get main.js]
                    HTTP::respond 200 content $contentValue "Cache-Control" "no-store"
                    table delete $apm_cookie
                    log local0. "on edgclient EC disconect, rendered main.js and delete entry $apm_cookie"
                }
                else {
                    #defaultmain js file for webvpn cases and tells browser to cache no-store
                    set contentValue [ifile get defaultmain.js]
                    HTTP::respond 200 content $contentValue "Cache-Control" "no-store"
                    table delete $apm_cookie
                    log local0. "on webtop disconnect, rendered defaultmain.js and delete entry $apm_cookie"
                }
            }
            else {
                #defaultmain js file for webvpn cases and tells browser to cache no-store
                set contentValue [ifile get defaultmain.js]
                log local0. "on disconect, rendered defaultmainjs"
                HTTP::respond 200 content $contentValue "Cache-Control" "no-store"
            }
        }
    }
}

when HTTP_RESPONSE {
    if {[info exists hangupReq]} {
        log local0. "Response for hangup on Access policy failure"
        if { [HTTP::header exists "Content-Length"] and [HTTP::header "Content-Length"] <= 1048576 } {
            HTTP::collect [HTTP::header Content-Length]
        } else {
            HTTP::collect 2097152 ;# 2MB
        }
    }
}

when HTTP_RESPONSE_DATA {
    if {[info exists hangupReq]} {
        unset hangupReq
        log local0. "hangup Response -- $redirect_port $state_param $session_id"
        set html_response2 "<html><head><title>Connecting to Edge client.</title><link rel=\"stylesheet\" href=\"/public/include/css/modern/framework.css\" /></head>"
        append html_response2 "<body><div id=\"apmui-container\" class=\"apmui-container\"><div class=\"apmui-root apmui-page-policy apmui-page-logout\">"
        append html_response2 "<header class=\"apmui-header\"><div class=\"apmui-header--content\"><picture class=\"apmui-header--logo\">"
        append html_response2 "<source media=\"(max-width: 586px)\" srcset=\"/public/images/modern/general/f5-logo-mobile.png\">"
        append html_response2 "<img src=\"/public/images/modern/general/f5-logo-desktop.png\" alt=\"F5 Logo\"></picture><span></span></div></header><main class=\"apmui-main\">"
        append html_response2 "<div class=\"apmui-content\"><h3 id=\"responseText\">Access was denied by the access policy. This may be due"
        append html_response2 "to a failure to meet access policy requirements.</h3></div></main><footer class=\"apmui-footer\">"
        append html_response2 "<div class=\"apmui-footer--content\" id=\"copyright\">This product is licensed from F5, Inc. (c) 1999-2024 F5, Inc. All rights reserved.</div></footer></div></div>"

        append html_response2 "<script> var F5_ST = null; var TIN = null; const url = new URL(window.location.href);"
        append html_response2 "const hostname = url.protocol + \"//\" + url.hostname + (url.port ? \":\" + url.port : \"\");"
        append html_response2 "const successText = \"Access was denied by the access policy. This may be due to a failure to meet access policy requirements. You can close this page now.\";"
        append html_response2 "function closeTab() \{setTimeout(() => \{window.close();\}, \"5000\");\}"
        append html_response2 "function sendReqToLocalhost(port, otcCode, state) \{const xhr = new XMLHttpRequest();xhr.open(\"GET\", \"http://localhost:\"+ port + \"?code=\" + otcCode + \"&f5_st=\" + F5_ST + \"&tin=\" + TIN + \"&host=\" + hostname + \"&state=\" + state, true);"
        append html_response2 "xhr.send(null);xhr.onreadystatechange = function () \{ if( xhr.readyState === XMLHttpRequest.DONE && xhr.status === 200) \{"
        append html_response2 "document.getElementById(\"responseText\").innerHTML= successText; \}else if (xhr.readyState === XMLHttpRequest.DONE && xhr.status !== 200) \{"
        append html_response2 "document.getElementById(\"responseText\").innerHTML=\"Something went wrong, Communication with server failed.\";\}\}\}"
        append html_response2 "sendReqToLocalhost($redirect_port, \"-1\", \"$state_param\"); </script></body></html>"
        set len [string length $html_response2]
        HTTP::payload replace 0 [HTTP::payload length] ""

        set html_response2 [binary format a* $html_response2]
        while true {
            if {$len <= 50000} {
                set chunk $len
            } else {
                set chunk 50000
            }
            set len [expr {$len - $chunk}]
            HTTP::payload replace 0 0 [string range $html_response2 $len [expr {$len + $chunk - 1}]]
            if {$len == 0} break;
        }
    }
}

when ACCESS_SESSION_STARTED {
    log local0. "Session Started."
    set session_id  [ACCESS::session sid]
    set landinguri [ACCESS::session data get -sid $session_id session.server.landinguri]
    log local0. "landing uri: $landinguri    and sessionid: $session_id"
    if { $landinguri contains "?EC=1&port=" } {
        ACCESS::session data set session.client.type "Standalone"
        log local0. "session modified and has uri: [HTTP::uri]"
    }
}
