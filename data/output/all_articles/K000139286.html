K000139286: Authentication tab is missing in VPE for RDG-RAP Access Policy type

Published

Updated

Source

Description

Authentication tab is missing in Visual Policy Editor (VPE) for RDG-RAP Access Policy type with AD/LDAP Query agents

 

Environment

  • BIG-IP APM v15.X
  • RDG-RAP Access Policy

 

 

 Recommended Actions

 

  • Use the TMSH CLI to configure AD/LDAP Query for the RDG-RAP Access Policy type.

 

Workaround

 

// Go into TMSH

 

create cli transaction
create apm policy agent ending-allow /Common/RDP_RAP_end_allow_ag { }
create apm policy agent ending-deny /Common/RDP_RAP_end_deny_ag { }
create apm policy agent aaa-active-directory /Common/RDP_RAP_act_active_directory_query_ag query-attrname add { cn displayName distinguishedName dn memberOf sAMAccountName sn } server AD_APMJS type query

 

create apm policy policy-item /Common/RDP_RAP_end_allow agents add { RDP_RAP_end_allow_ag { type ending-allow } } caption Allow color 1 item-type ending

create apm policy policy-item /Common/RDP_RAP_end_deny agents add { RDP_RAP_end_deny_ag { type ending-deny } } caption Deny color 2 item-type ending

create apm policy policy-item /Common/RDP_RAP_ent { caption Start color 1 rules { { caption fallback next-item /Common/RDP_RAP_act_active_directory_query } } }

create apm policy policy-item /Common/RDP_RAP_act_active_directory_query agents add { /Common/RDP_RAP_act_active_directory_query_ag { type aaa-active-directory } } caption AD_Query color 1 item-type action rules { { caption fallback next-item /Common/RDP_RAP_end_allow } }

 

create apm policy access-policy RDP_RAP default-ending /Common/RDP_RAP_end_allow items add { /Common/RDP_RAP_end_allow {} /Common/RDP_RAP_end_deny {} /Common/RDP_RAP_act_active_directory_query {} /Common/RDP_RAP_ent {} } start-item /Common/RDP_RAP_ent

 

create apm profile access RDP_RAP accept-languages add { en } access-policy RDP_RAP type rdg-rap

 

submit cli transaction
