Description

You may notice that passphrase of f5_api_com.key is changed when generating the qkview.

For example :

[root@LTM1.example.com:Active:Standalone] config # tmsh list sys file ssl-key f5_api_com.key
sys file ssl-key f5_api_com.key {
    checksum SHA1:3306:e642cec9c7cab44db586bf8a6cce3e7e1423b628
    create-time 2020-10-15:18:31:39
    created-by root
    key-size 4096
    last-update-time 2020-10-15:18:31:39
    mode 33184
    passphrase $M$On$I+0rCEBVa7KKCBRzTHYje48Nb/1th3pqFRuMpdPlWc4iPFBAxhYbG2e16+yA9mZnJ6/+Glc96KFSG8mPrQzCgrS/DKGR+W2zKioIvgRYTTw=
    revision 1
    security-type password
    size 3306
    source-path file:///config/ssl/ssl.key/f5_api_com.key
    updated-by root
}

Generating qkview

[root@LTM1.example.com:Active:Standalone] config # qkview
Gathering System Diagnostics: Please wait ...
Diagnostic information has been saved in:
/var/tmp/BIG-IP.local.qkview
Please send this file to F5 support.

[root@LTM1.example.com:Active:Standalone] config # tmsh list sys file ssl-key f5_api_com.key
sys file ssl-key f5_api_com.key {
    checksum SHA1:3306:e642cec9c7cab44db586bf8a6cce3e7e1423b628

    created-by root
    key-size 4096
    last-update-time 2020-10-15:18:31:39
    mode 33184
    passphrase $M$V3$dWfeg1DrZp1oKeetcksNX48Nb/1th3pqFRuMpdPlWc4iPFBAxhYbG2e16+yA9mZnJ6/+Glc96KFSG8mPrQzCgrS/DKGR+W2zKioIvgRYTTw=              revision 1
    security-type password
    size 3306

    updated-by root
}

Environment

  • BIG-IP
  • Qkview generation

Cause
This is expected behavior of Secure Vault feature.
