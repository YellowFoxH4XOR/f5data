Description

Due to specific needs, it's possible to assign the access-profile logic from irule only. Then the irule is assigned under virtual-server. The configuration example:

root@(bigip)(cfg-sync Standalone)(Active)(/Common)(tmos)# list ltm virtual vs
ltm virtual vs {  #### we don't assign the access-profile /Common/ap_test under Virtual server
    destination 10.1.1.1:https
    ip-protocol tcp
    last-modified-time 2022-04-05:02:34:05
    mask 255.255.255.255
    pool pool_dummy
    profiles {
        clientssl {
            context clientside
        }
        http { }
        tcp { }
    }
    rules {
        irule_article
    }
    serverssl-use-sni disabled
    source 0.0.0.0/0
    translate-address enabled
    translate-port enabled
    vs-index 4
}

root@(bigip)(cfg-sync Standalone)(Active)(/Common)(tmos)# list ltm rule irule_article
ltm rule irule_article {
when HTTP_REQUEST {
  # Set session authentication, idle and lifetime timeouts
  set auth_timeout 300
  set apm_timeout 600
  set apm_lifetime 1800
  # Set Virtual server name
  set vs_name [lindex [split [virtual name] "/"] end]  
  # Set the APM profile used
  set apm_profile "/Common/ap_test"
  set apm_sid ""
  # Here can be different checks whether the session already exists. This code portion is omitted.
  if { $apm_sid eq "" } {
    # create a new APM session for this request
    set apm_sid [ACCESS::session create -timeout $apm_timeout -lifetime $apm_lifetime]
  }
  # now evaluate APM policy for this request
  ACCESS::policy evaluate -sid $apm_sid -profile $apm_profile
  # Check the outcome of the APM policy evaluation and do some action based on evaluation result
  switch -- [ACCESS::policy result -sid $apm_sid] {
    "allow" {
      log local0. "we are in allow section"
  }
    "deny" {
        log local0. "we are in deny section"
      }
    default {
        log local0. "we are in default section"
      }
  }
  }
}
root@(bigip)(cfg-sync Standalone)(Active)(/Common)(tmos)# list apm profile access /Common/ap_test
apm profile access ap_test {
    accept-languages { en }
    access-policy ap_test
    app-service none
    customization-group ap_test_logout
    customization-key 77256eeccf57cf00e5207b13ff0cdbb5
    default-language en
    domain-cookie none
    eps-group ap_test_eps
    errormap-group ap_test_errormap
    exchange-profile none
    framework-installation-group ap_test_framework_installation
    general-ui-group ap_test_general_ui
    generation 4
    generation-action noop
    httponly-cookie false
    log-settings {
        default-log-setting
    }
    logout-uri-include none
    logout-uri-timeout 5
    modified-since-last-policy-sync true
    named-scope none
    oauth-profile none
    persistent-cookie false
    scope profile
    secure-cookie true
    sso-name none
    type all
    user-identity-method http
}

As the result of such configuration, there are following symptoms can be observed:
  1. The apm session can't be deleted in GUI, but can be deleted in CLI. The session checkmark in GUI is greyed out. When we move the mouse to the checkmark, we can see the following statement:

    "This item can not be deleted here because the device is in standby or the profile type is unknown".

  2. The session created by this irule is seen in GUI "Access  ››  Overview : Active Sessions, the column 'Session Type'  as 'irule'
  3. The session count and statistical information is not seen in tmsh command outputs. For example, in 'tmsh list apm profile access' .  

Environment

  • apm
  • access-profile
  • irule

Cause

Such limitations are expected and by current design. There is an existing ID1089957 opened for tracking these limitations.

Recommended Actions

Should you experience the same symptoms, then open the ticket with F5 Support and request to add your case to ID1089957.
