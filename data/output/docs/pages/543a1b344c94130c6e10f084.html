equals

Description

Tests if one string equals another string.

Syntax

<string1> equals <string2>

<string1> equals <string2>

  • Tests if string1 equals string2.

Examples

when HTTP_REQUEST {
   if { [HTTP::uri] equals "/login" } {
      pool login_pool
   } else {
      pool default_pool
   }
}


when CLIENT_ACCEPTED {
   if { class match [IP::client_addr] equals aol] } {
      pool aol_pool
   } else {
      pool all_pool
   }
}
